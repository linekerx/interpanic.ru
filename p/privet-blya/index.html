    1. Создание исходного кода полезной нагрузки

    cat << 'EOF' > payload.asm
    bits 64
    global _start

    section .text
    _start:
    mov rax, 60
    mov rdi, 42
    syscall
    EOF

    2. Создание скрипта-упаковщика (packer.py)

    cat << 'EOF' > packer.py
    import sys

    KEY = b"K9#m$P2!vL8@zX5*qW7&fB3%dN$z9"

    def encrypt_payload(input_file, output_file):
    with open(input_file, "rb") as f:
    plaintext = f.read()

    ciphertext = bytearray()
    key_len = len(KEY)
    for i, b in enumerate(plaintext):
    ciphertext.append(b ^ KEY[i % key_len])

    with open(output_file, "w") as f:
    f.write("; Auto-generated encrypted payload\n")
    f.write("section .data\n")
    f.write("global encrypted_payload\n")
    f.write("global payload_size\n\n")
    f.write("encrypted_payload:\n db ")

    hex_bytes = [f"0x{b:02x}" for b in ciphertext]
    for i in range(0, len(hex_bytes), 12):
    f.write(", ".join(hex_bytes[i:i+12]) + "\n db ")

    f.seek(f.tell() - 6)
    f.write(f"\npayload_size dq {len(ciphertext)}\n")

    if __name__ == "__main__":
    encrypt_payload(sys.argv[1], sys.argv[2])
    EOF

    3. Создание исходного кода загрузчика

    cat << 'EOF' > loader.asm
    bits 64
    global _start

    extern encrypted_payload
    extern payload_size

    section .data
    key db "K9#m$P2!vL8@zX5*qW7&fB3%dN$z9", 0
    key_len equ 28

    section .text
    _start:
    mov rsi, encrypted_payload
    mov rcx, [payload_size]
    mov rdi, rsi
    xor r8, r8

    decrypt_loop:
    cmp rcx, 0
    je make_executable

    mov al, byte [rdi]
    mov bl, byte [key + r8]
    xor al, bl
    mov byte [rdi], al

    inc rdi
    inc r8
    xor rdx, rdx
    mov rax, r8
    mov rbx, key_len
    div rbx
    mov r8, rdx

    dec rcx
    jmp decrypt_loop

    make_executable:
    mov rdi, encrypted_payload
    mov rax, rdi
    and rax, 0xFFFFFFFFFFFFF000

    mov rsi, [payload_size]
    add rsi, 4095
    and rsi, 0xFFFFFFFFFFFFF000

    mov rdx, 7
    mov rax, 10
    syscall

    test rax, rax
    js exit_error

    jmp encrypted_payload

    exit_error:
    mov rax, 60
    mov rdi, 1
    syscall
    EOF

    4. Установка зависимостей (для Debian / Ubuntu / Astra / WSL)

    sudo apt update && sudo apt install -y nasm binutils python3

    5. Сборка полезной нагрузки

    nasm -f elf64 payload.asm -o payload.o
    ld payload.o -o payload

    6. Шифрование и генерация payload.inc

    python3 packer.py payload payload.inc

    7. Сборка загрузчика

    nasm -f elf64 loader.asm -o loader.o
    ld loader.o -o memory_loader

    8. Запуск и проверка кода возврата

    ./memory_loader
    echo $?
